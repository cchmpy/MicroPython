2.2 
	简单自动的安装方法
	wsl --install
	手动安装WSL的步骤
    (1) dism.exe /online /enable-feature /featurename:Microsoft-Windows-Subsystem-Linux /all /norestart
    (2) dism.exe /online /enable-feature /featurename:VirtualMachinePlatform /all /norestart
    (4) wsl --set-default-version 2    
    卸载wsl
    (1) wsl --unregister Ubuntu
	
2.3	下载MicroPython源代码
	cd
	tar xvJf micropython-xxx.tar.xz 

2.4 配置编译环境命令
	sudo apt update
	sudo apt install build-essential gcc-mingw-w64 cmake 

2.5 克隆软件包
	cd
	git clone https://gitee.com/EspressifSystems/esp-gitee-tools.git  
	git clone https://gitee.com/EspressifSystems/esp-idf.git
    
	安装辅助环境和库
	python3 --version
	sudo apt install python3.12-venv
	sudo apt-get install libusb-1.0

    切换到特定的ESP-IDF标签（版本）、更新子模块、安装ESP-IDF工具链
	cd ~/esp-idf
	checkout v5.5.1
	cd ~/esp-gitee-tools
	./submodule-update.sh ~/esp-idf/ 
	./install.sh ~/esp-idf
     
    设置环境变量
    source ~/esp-idf/export.sh

2.6 编译生成mpy-cross
	cd ~/micropython
	make -C mpy-cross

    关于交叉编译器
	cd ~/micropython/mpy-cross/build
	./mpy-cross main.py -march=xtensawin
        
    mpy-cross的用法
	./mpy-cross --help
	./mpy-cross 1.py -march=xtensawin -X emit=bytecode 
	./mpy-cross 2.py -march=xtensawin -X emit=native
	./mpy-cross 3.py -march=xtensawin -X emit=viper

    Windows系统下使用mpy-cross
	pip install mpy-cross
	mpy-cross --version
	mpy-cross --help
	pip install --upgrade mpy-cross
	mpy-cross 1.py -march=xtensawin -X emit=bytecode

    2.7.1 编译默认固件

		python ~/esp-idf/components/partition_table/gen_esp32part.py partitions-16MiB-custom.csv custom.bin

	查看固件文件（firmware.bin）使用的分区表信息
